1️⃣  CREATE DIRECTORIES (2 min)
   mkdir -p barter-app/backend/src/{config,models,middleware,routes}
   mkdir -p barter-app/frontend/src/{screens,components,services,context}

2️⃣  COPY BACKEND FILES (3 min)
   • BACKEND_INDEX_JS.txt → backend/src/index.js
   • BACKEND_JWT_CONFIG.txt → backend/src/config/jwt.js
   • BACKEND_AUTH_MIDDLEWARE.txt → backend/src/middleware/auth.js
   • BACKEND_*_MODEL.txt files → backend/src/models/
   • BACKEND_*_ROUTES.txt files → backend/src/routes/

3️⃣  COPY FRONTEND FILES (2 min)
   • FRONTEND_API_SERVICE.txt → frontend/src/services/api.ts
   • FRONTEND_AUTH_CONTEXT.txt → frontend/src/context/AuthContext.tsx
   • FRONTEND_*_SCREEN.txt files → frontend/src/screens/

4️⃣  INSTALL DEPENDENCIES (5 min)
   cd backend
   npm init -y && npm install express mongoose dotenv bcryptjs jsonwebtoken cors express-validator multer nodemon
   cd ../frontend
   npx create-expo-app . && npm install

5️⃣  RUN & TEST (3 min)
   # Terminal 1: Backend
   cd backend && npm run dev
   
   # Terminal 2: Frontend
   cd frontend && npm start
